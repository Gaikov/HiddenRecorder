cmake_minimum_required(VERSION 3.24)
project(HiddenRecorderMedia LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(GST_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.tools/gstreamer" CACHE PATH "GStreamer MSVC SDK")
add_library(HiddenRecorder.Media SHARED media.cpp)
target_compile_definitions(HiddenRecorder.Media PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(HiddenRecorder.Media PRIVATE "${GST_ROOT}/include/gstreamer-1.0" "${GST_ROOT}/include/glib-2.0" "${GST_ROOT}/lib/glib-2.0/include")
target_link_directories(HiddenRecorder.Media PRIVATE "${GST_ROOT}/lib")
target_link_libraries(HiddenRecorder.Media PRIVATE gstreamer-1.0 gstapp-1.0 gstvideo-1.0 gobject-2.0 glib-2.0)
target_compile_options(HiddenRecorder.Media PRIVATE /W4 /EHsc /utf-8)
add_executable(media-smoke smoke.cpp)
target_link_libraries(media-smoke PRIVATE HiddenRecorder.Media)
add_executable(ring-tests ring_tests.cpp)
target_compile_definitions(ring-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(ring-tests PRIVATE "${GST_ROOT}/include/gstreamer-1.0" "${GST_ROOT}/include/glib-2.0" "${GST_ROOT}/lib/glib-2.0/include")
target_link_directories(ring-tests PRIVATE "${GST_ROOT}/lib")
target_link_libraries(ring-tests PRIVATE gstreamer-1.0 gstapp-1.0 gstvideo-1.0 gobject-2.0 glib-2.0)
target_compile_options(ring-tests PRIVATE /W4 /EHsc /utf-8)
enable_testing()
add_test(NAME ring-tests COMMAND ring-tests)
add_executable(mf-seek-check mf_seek_check.cpp)
target_compile_definitions(mf-seek-check PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(mf-seek-check PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(mf-seek-check PRIVATE mfplat mfreadwrite mfuuid ole32 propsys)
